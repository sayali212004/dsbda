# Start Hive
hive

# Create external table mapped to HBase table
CREATE EXTERNAL TABLE stud(
    key string,
    name string,
    age string
)
STORED BY 'org.apache.hadoop.hive.hbase.HBaseStorageHandler'
WITH SERDEPROPERTIES (
    "hbase.columns.mapping" = ":key,info:name,info:age"
)
TBLPROPERTIES ("hbase.table.name" = "student");

# Query data
SELECT * FROM stud;

# Insert new record (will be stored in HBase)
INSERT INTO stud VALUES ('3','Ankit','20');
